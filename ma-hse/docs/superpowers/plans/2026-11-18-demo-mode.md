# Modo demo local: plano de implementação

> **Para agentes:** SUB-SKILL OBRIGATÓRIO: usar superpowers:subagent-driven-development (recomendado) ou superpowers:executing-plans para executar este plano tarefa a tarefa. Os passos usam checkboxes (`- [ ]`) para acompanhamento.

**Objetivo:** permitir correr `npm run demo` no portátil e ter uma instalação MAx Safety isolada, com dados fictícios, um guião de apresentação e guiões de formação (N3, N1, N0) em PT e EN.

**Arquitetura:**
- **Ambiente:** a demo é o mesmo código a correr com `.env.demo` contra um compose próprio (postgres, redis e minio noutras portas).
- **`DEMO_MODE`:** só liga interruptores. Faz uma guarda de segurança, suprime o envio de emails, mostra a faixa e as contas demo, e ativa o reset e os guiões.
- **Dados:** gerados de forma determinística por `prisma/demo-seed.ts`, com datas relativas ao dia em que corre.
- **Guiões:** estendem o motor de onboarding existente (`driver.js`) com passos de "tarefa" e guiões nomeados.

**Stack:** Next.js 16 (App Router), Prisma 6.19, PostgreSQL 16 + pgvector, BullMQ/Redis, MinIO, NextAuth v4, Zod 4, Vitest + Testing Library, `driver.js` 1.8.

**Spec:** `docs/superpowers/specs/2026-11-18-demo-mode-design.md`

```ts
export type DemoScriptId = "demo-presentation" | "demo-training-n3" | "demo-training-n1" | "demo-training-n0";

export const DEMO_PASSWORD = "Demo2026!";

export const DEMO_PLANTS = [
  { code: "demo-lis", name: "Demo Lisboa", timezone: "Europe/Lisbon", defaultLanguage: "pt" },
  { code: "demo-opo", name: "Demo Porto", timezone: "Europe/Lisbon", defaultLanguage: "pt" },
] as const;

export const DEMO_ACCOUNTS = [
  { email: "demo.apresentacao@maxsafety.demo", name: "Apresentação Demo", role: "N1_CORPORATE", plantCode: null, script: "demo-presentation" },
  { email: "demo.admin@maxsafety.demo", name: "Admin Demo", role: "N0_ADMIN", plantCode: null, script: "demo-training-n0" },
  { email: "demo.corporate@maxsafety.demo", name: "Corporate Demo", role: "N1_CORPORATE", plantCode: null, script: "demo-training-n1" },
  { email: "demo.seguranca@maxsafety.demo", name: "Segurança Demo", role: "N3_SAFETY", plantCode: "demo-lis", script: "demo-training-n3" },
] as const satisfies ReadonlyArray<{
  email: string; name: string; role: "N0_ADMIN" | "N1_CORPORATE" | "N3_SAFETY"; plantCode: string | null; script: DemoScriptId;
}>;

const LOCAL_HOSTS = new Set(["localhost", "127.0.0.1", "::1", "[::1]"]);

export function assertDemoModeSafe(input: { DEMO_MODE: boolean; APP_ENV: string; DATABASE_URL: string }) {
  if (!input.DEMO_MODE) return;
  if (input.APP_ENV === "production") {
    throw new Error("DEMO_MODE não pode ser usado em produção (APP_ENV=production).");
  }
  let url: URL;
  try {
    url = new URL(input.DATABASE_URL);
  } catch {
    throw new Error("DEMO_MODE: DATABASE_URL inválido.");
  }
  if (!LOCAL_HOSTS.has(url.hostname)) {
    throw new Error(`DEMO_MODE: a base de dados tem de ser local (recebido host "${url.hostname}").`);
  }
  if (url.pathname.replace(/^\//, "") !== "ehs_demo") {
    throw new Error("DEMO_MODE: a base de dados tem de se chamar ehs_demo.");
  }
}

export function isDemoMode() {
  return process.env.DEMO_MODE === "true";
}

export function getDemoScriptForEmail(email: string | null | undefined): DemoScriptId | null {
  const normalized = email?.trim().toLowerCase();
  return DEMO_ACCOUNTS.find((account) => account.email === normalized)?.script ?? null;
}
```

```js
import fs from "node:fs";
import path from "node:path";
import { parse } from "dotenv";

/** Demo env values. They override the parent env, including empty values, so dev secrets never leak in. */
export function loadDemoEnv() {
  const file = path.join(process.cwd(), ".env.demo");
  if (!fs.existsSync(file)) throw new Error(".env.demo não encontrado.");
  const demo = parse(fs.readFileSync(file));
  return { ...process.env, ...demo };
}
```

```ts
import crypto from "node:crypto";

export function createRandom(seed: string) {
  let state = crypto.createHash("sha1").update(seed).digest().readUInt32LE(0);
  const next = () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
  return {
    next,
    int: (min: number, max: number) => min + Math.floor(next() * (max - min + 1)),
    pick: <T,>(items: readonly T[]) => items[Math.floor(next() * items.length)],
    chance: (p: number) => next() < p,
  };
}
```

```ts
import type { OnboardingTaskCondition } from "@/components/onboarding/onboarding-types";

export function routePatternMatches(pattern: string, pathname: string) {
  const escaped = pattern
    .split("/")
    .map((segment) => (segment === "**" ? "§§" : segment === "*" ? "§" : segment.replace(/[.+?^${}()|[\]\\]/g, "\\$&")))
    .join("/")
    .replace("/§§", "(?:/.*)?")
    .replaceAll("§", "[^/]+");
  return new RegExp(`^${escaped}$`).test(pathname);
}

export function isTaskComplete(condition: OnboardingTaskCondition, pathname: string, doc: Pick<Document, "querySelector">) {
  if (condition.route && !routePatternMatches(condition.route, pathname)) return false;
  if (condition.element && !doc.querySelector(condition.element)) return false;
  return Boolean(condition.route || condition.element);
}
```
